require("dotenv").config();
const express = require("express");
const cors = require("cors");
const nodemailer = require("nodemailer");
const bcrypt = require('bcrypt');
const validator = require('validator');
const jwt = require('jsonwebtoken');
const {rateLimit,ipKeyGenerator} = require('express-rate-limit');

const PORT = process.env.PORT;

const app = express();

app.use(express.json());
app.use(cors());

const users = [];


const loginLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max:5,
  message:{error:"Demasiados intentos, intenta más tarde"},
  keyGenerator: (req) => {
    return ipKeyGenerator(req) + "-" + req.body.correo;
  },

  skipSuccessfulRequests:true,
})

const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: process.env.SMTP_PORT,
  auth: {
    user: process.env.EMAIL_USER,
    pass: process.env.EMAIL_PASS,
  },
});

const signUp = () => {
  app.post("/signup",async(req,res) => {
    try {
      const {nombre,contraseña,correo} = req.body;

      if (!nombre || nombre.length < 3) {
        return res.status(400).json({error:"Nombre inválido"});
      }

      const isEmailValid = validator.isEmail(correo, 
        {allow_utf8_local_part:false,
          allow_ip_domain:false
        }
      );

      if (!isEmailValid) {
        return res.status(400).json({error:"Correo inválido"});
      }

      const isValidPassword = validator.isStrongPassword(contraseña, {
        minLength:8,
        minLowercase:1,
        minUppercase:1,
        minNumbers:1,
        minSymbols:1,
      })

      if (!isValidPassword) {
        return res.status(400).json({error:"Contraseña no segura. Debe tener al menos 8 caracteres, una letra mayúscula, una letra minúscula, un número y un símbolo."})
      }

      const userExist = users.find(u => u.correo === correo);
      if (userExist) {
        return res.status(400).json({error:"El usuario ya existe"});
      }

      const token = jwt.sign(
        {correo},
        process.env.JWT_SECRET,
        {expiresIn:"15m"},
      )
    
      const link =  `http://localhost:3000/verify?token=${token}`;

      const hanshedPassword = await bcrypt.hash(contraseña,10);

      users.push({
        id: users.length + 1,
        nombre,
        correo,
        contraseña:hanshedPassword,
        verified:false
      })

      const emailOptions = {
        from:process.env.EMAIL_USER,
        to:correo,
        subject:"Bienvenido a nuestra página web",
        html:"<h1>Hola Juan, Gracias por haberte registrado en mi página</h1> Este email es de prueba"
      }

      transporter.sendMail(emailOptions)
      return res.status(200).json({
        message:"Usuario creado exitosamente"
      });
    } catch (error) {
      console.error(`Error: ${error}`);
      return res.status(500).json({error:"Error interno del servidor"});
    }
  })

  app.get("/verify",(req,res) => {
    const {token} = req.query;

    try {
      const decoded = jwt.verify(token,process.env.JWT_SECRET);

      const user = users.find(u => u.correo === decoded.correo);

      if (!user) {
        return res.status(404).json({error:"Usuario no encontrado"});
      }

      user.verified = true;

      res.send("Correo verificado exitosamente");
    } catch (error) {
      return res.status(400).json({error:"Token inválido o expirado"});
    }

    res.send("Correo verificado exitosamente");
  })
}

const signIn = () => {
  app.post("/signin",async(req,res) => {
    try {
      const {correo, contraseña} = req.body;

      if (!validator.isEmail(correo) || !contraseña) {
        return res.status(400).json({error:"Credenciales inválidas"});
      }

      const user = users.find(u => u.correo === correo);
        if (!user) {
          return res.status(404).json({error:"Usuario no encontrado"});
      }

      const isMatch = await bcrypt.compare(contraseña,user.contraseña);

      if (!isMatch) {
      return res.status(401).json({error:"Contraseña incorrecta"});
      }

      const accessToken = jwt.sign(
        {correo:user.correo},
        process.env.JWT_SECRET,
        {expiresIn:"1h"}
      );

      const refreshToken = jwt.sign(
        {correo:user.correo},
        process.env.JWT_REFRESH_SECRET,
        {expiresIn:"7d"}
      )

      return res.status(200).json({
        message:"Inicio de sesión exitoso",
        accessToken,
        refreshToken
      });
    } catch (error) {
      console.error("Error:",error);
      return res.status(500).json({error:"Error interno del servidor"});
    }
  })
}

const getToken = (req,res,next) => {
    const authHeader = req.headers.authorization;

      if (!authHeader) {
          return res.status(401).json({error:"Access token requerido"});
      }

      const token = authHeader.split(" ")[1];

      if (!token) {
          return res.status(401).json({error:"Token inválido"});
      }

      try {
          const decoded = jwt.verify(
            token,
            process.env.JWT_SECRET
          );

          req.user = decoded;
          return next();
        } catch (error) {
          return res.status(403).json({error:"Accesso denegado"});
    }
}

const validateToken = () => {
    app.get("/validate", getToken,(req,res) => {
      return res.json({
        message:"Token válido",
        user:req.user
      })
    })
}

const api = () => {
  app.listen(PORT,() => {
    console.log(`Servidor corriendo en puerto ${PORT}`);
  })
}

const routers = () => {
  signUp();
  signIn();
  validateToken();
}

routers();
api();
